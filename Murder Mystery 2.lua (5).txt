local LocalPlayer = Players.LocalPlayer
local ESPActive = false
local ESPs = {}

local function ClearESP()
    for _, v in pairs(ESPs) do
        if v then
            v:Destroy()
        end
    end
    table.clear(ESPs)
end

local function GetPart(obj)
    if obj:IsA("BasePart") then
        return obj
    end

    return obj:FindFirstChildWhichIsA("BasePart", true)
end

local function CreateESP(obj)
    if ESPs[obj] then
        return
    end

    local Part = GetPart(obj)
    if not Part then
        return
    end

    local Billboard = Instance.new("BillboardGui")
    Billboard.Name = "GunDropESP"
    Billboard.Adornee = Part
    Billboard.Size = UDim2.fromOffset(180, 35)
    Billboard.StudsOffset = Vector3.new(0, 3, 0)
    Billboard.AlwaysOnTop = true
    Billboard.Parent = Part

    local Label = Instance.new("TextLabel")
    Label.Size = UDim2.fromScale(1, 1)
    Label.BackgroundTransparency = 1
    Label.TextColor3 = Color3.new(1, 1, 1)
    Label.TextStrokeTransparency = 0
    Label.TextSize = 14
    Label.Font = Enum.Font.SourceSansBold
    Label.Parent = Billboard

    ESPs[obj] = Billboard
end

local function Scan()
    if not ESPActive then
        return
    end

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj.Name == "GunDrop" then
            CreateESP(obj)
        end
    end
end

Main:Toggle("ESP GunDrop", false, function(Value)
    ESPActive = Value
    if ESPActive then
        Scan()
    else
        ClearESP()
    end
end)

workspace.DescendantAdded:Connect(function(obj)
    if ESPActive and obj.Name == "GunDrop" then

    end
end)

workspace.DescendantRemoving:Connect(function(obj)

local ESPObjects = {}

local function CreateESP(player, color)
    if player == LocalPlayer then return end
    local Character = player.Character
    if not Character then return end
    local Root = Character:FindFirstChild("HumanoidRootPart")
    if not Root then return end

        ESPObjects[player]:Destroy()

    end

    Billboard.Name = "NameEsp"
    Billboard.ExtentsOffset = Vector3.new(0, 1, 0)
    Billboard.Size = UDim2.new(1, 200, 1, 30)

    Label.TextSize = 14
    Label.TextWrapped = true
    Label.Size = UDim2.new(1, 0, 1, 0)
    Label.TextYAlignment = Enum.TextYAlignment.Top
    Label.BackgroundTransparency = 1
    Label.TextStrokeTransparency = 0.5
    Label.TextColor3 = color
    Label.Parent = Billboard
    
    ESPObjects[player] = Billboard
end

local function GetPlayerRole(player)
    local Character = player.Character
    if not Character then return "Innocent" end
    local backpack = player:FindFirstChild("Backpack")
    
    local bodyColors = Character:FindFirstChild("Body Colors")
    if bodyColors then
        local leftArmColor = bodyColors:FindFirstChild("LeftArmColor3")
        if leftArmColor and leftArmColor.Value == Color3.fromRGB(80, 109, 84) then
            return "Zombie"
        end
    end

    if Character:FindFirstChild("Knife") or (backpack and backpack:FindFirstChild("Knife")) then
        return "Murderer"
    elseif Character:FindFirstChild("Gun") or (backpack and backpack:FindFirstChild("Gun")) then
        return "Sheriff"
    elseif Character:FindFirstChild("Zombie") or (backpack and backpack:FindFirstChild("Zombie")) then
        return "Zombie"
    elseif Character:FindFirstChild("Survivor") or (backpack and backpack:FindFirstChild("Survivor")) then
        return "Survivor"
    elseif Character:FindFirstChild("Freezer") or (backpack and backpack:FindFirstChild("Freezer")) then
        return "Freezer"
    elseif Character:FindFirstChild("Runner") or (backpack and backpack:FindFirstChild("Runner")) then
        return "Runner"
    end
    
    return "Innocent"
end

local function UpdateESP()
    if not innocentESPToggled and not sheriffESPToggled and not murdererESPToggled 
       and not zombieESPToggled and not survivorESPToggled and not freezerESPToggled and not runnerESPToggled then
        for player, Billboard in pairs(ESPObjects) do
            if Billboard then Billboard:Destroy() end
        end
        ESPObjects = {}
        return
    end
    
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local Character = player.Character
            local Root = Character and Character:FindFirstChild("HumanoidRootPart")
            if Root then
                local role = GetPlayerRole(player)
                local shouldShow = false
                local color = Color3.fromRGB(255, 255, 255)
                
                if role == "Innocent" and innocentESPToggled then
                    shouldShow = true
                    color = Color3.fromRGB(0, 255, 100)
                elseif role == "Sheriff" and sheriffESPToggled then
                    shouldShow = true
                    color = Color3.fromRGB(0, 150, 255)
                elseif role == "Murderer" and murdererESPToggled then
                    shouldShow = true
                    color = Color3.fromRGB(255, 50, 50)
                elseif role == "Zombie" and zombieESPToggled then
                    shouldShow = true
                    color = Color3.fromRGB(25, 172, 0)
                elseif role == "Survivor" and survivorESPToggled then
                    shouldShow = true
                    color = Color3.fromRGB(43, 154, 238)
                elseif role == "Freezer" and freezerESPToggled then
                    shouldShow = true
                    color = Color3.fromRGB(150, 220, 250)
                elseif role == "Runner" and runnerESPToggled then
                    shouldShow = true
                    color = Color3.fromRGB(0, 200, 100)
                end
                
                if shouldShow then
                    if not ESPObjects[player] then
                        CreateESP(player, color)
                    end
                    local Billboard = ESPObjects[player]
                    if Billboard then
                        local Label = Billboard:FindFirstChild("TextLabel")
                        if Label then
                            local distance = 0
                            if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Head") then
                                distance = math.floor((LocalPlayer.Character.Head.Position - Root.Position).Magnitude / 3)
                            end
                            Label.Text = player.Name .. " [" .. tostring(distance) .. "m]"
                            Label.TextColor3 = color
                        end
                    end
                elseif ESPObjects[player] then
                    ESPObjects[player]:Destroy()
                    ESPObjects[player] = nil
                end
            elseif ESPObjects[player] then
                ESPObjects[player]:Destroy()
                ESPObjects[player] = nil
            end
        end
    end
end

local function GetTimerObject()
    local found
    for _, v in ipairs(workspace:GetDescendants()) do
        if v:GetAttribute("Time") and v:FindFirstChild("SurfaceGui") then
            found = v
            break
        end
    end
    return found
end

local function UpdateTimer()
    local timerObj = GetTimerObject()
    if timerObj then
        local timeLeft = timerObj:GetAttribute("Time") or -1
        if timeLeft > 0 then
            local minutes = math.floor(timeLeft / 60)
            local seconds = timeLeft - (minutes * 60)
            TimerLabel.Text = (minutes > 0 and minutes .. "m " or "") .. seconds .. "s"
            TimerLabel.Visible = true
        else
            TimerLabel.Visible = false
        end
    else
        TimerLabel.Visible = false
    end
end

local function RealFling(TargetPlayer)
    local MyChar = LocalPlayer.Character
    local TargetChar = TargetPlayer.Character
    if not MyChar or not TargetChar then return end
    
    local MyHRP = MyChar:FindFirstChild("HumanoidRootPart")
    local MyHum = MyChar:FindFirstChildOfClass("Humanoid")
    local TargetHRP = TargetChar:FindFirstChild("HumanoidRootPart")
    local TargetHum = TargetChar:FindFirstChildOfClass("Humanoid")
    
    if not MyHRP or not MyHum or not TargetHRP or not TargetHum then return end

    local OldPos = MyHRP.CFrame
    
    local function ApplyFling(Pos, Rot)
        MyHRP.CFrame = CFrame.new(Pos.Position) * Rot
        pcall(function()
            MyChar:SetPrimaryPartCFrame(CFrame.new(Pos.Position) * Rot)
        end)
        MyHRP.Velocity = Vector3.new(90000000, 900000000, 90000000)
        MyHRP.RotVelocity = Vector3.new(900000000, 900000000, 900000000)
    end

    local BodyVelocity = Instance.new("BodyVelocity")
    BodyVelocity.Name = "RealFlingVel"
    BodyVelocity.Parent = MyHRP
    BodyVelocity.Velocity = Vector3.new(900000000, 900000000, 900000000)
    BodyVelocity.MaxForce = Vector3.new(math.huge, math.huge, math.huge)

    MyHum:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

    local StartTime = tick()
    local Angle = 0
    
    while tick() - StartTime < 1.5 do
        if TargetHRP.Velocity.Magnitude < 50 then
            Angle = Angle + 100
            ApplyFling(TargetHRP, CFrame.new(0, 1.5, 0) + TargetHum.MoveDirection * TargetHRP.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
            task.wait()
            ApplyFling(TargetHRP, CFrame.new(0, -1.5, 0) + TargetHum.MoveDirection * TargetHRP.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
            task.wait()
            ApplyFling(TargetHRP, CFrame.new(2.25, 1.5, -2.25) + TargetHum.MoveDirection * TargetHRP.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
            task.wait()
        else
            ApplyFling(TargetHRP, CFrame.new(0, 1.5, TargetHum.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
            task.wait()
            ApplyFling(TargetHRP, CFrame.new(0, -1.5, -TargetHum.WalkSpeed), CFrame.Angles(0, 0, 0))
            task.wait()
            ApplyFling(TargetHRP, CFrame.new(0, 1.5, TargetHRP.Velocity.Magnitude / 1.25), CFrame.Angles(math.rad(90), 0, 0))
            task.wait()
        end
        
        if TargetHRP.Velocity.Magnitude > 500 or TargetChar.Parent ~= Workspace or TargetHum.Sit or MyHum.Health <= 0 then
            break
        end
    end

    if BodyVelocity and BodyVelocity.Parent then
        BodyVelocity:Destroy()
    end
    MyHum:SetStateEnabled(Enum.HumanoidStateType.Seated, true)

    if OldPos and MyHum.Health > 0 then
        MyHRP.CFrame = OldPos
    end
end

                if hrp and hum and hum.Health > 0 then
                    local isFull = false

                    if candyGui and candyGui:FindFirstChild("Lobby") then
                        local coinsLabel = candyGui.Lobby.Dock.CoinBags.Container.Candy.CurrencyFrame.Icon.Coins

                            local candyCount = tonumber(string.match(coinsLabel.Text, "%d+")) or 0

                                    dist = d
                                    nearest = v
                                end
                            end
                        end

                            if dist > 1000 then

                                hrp.AssemblyLinearVelocity = Vector3.zero

                                local currentSpeed = flySpeed
                                if dist <= 200 then
                                    currentSpeed = flySpeed
                                end

                                if flyMode == "down" then
                                    targetPos = targetPos - Vector3.new(0, 3, 0)
                                end
                                local dir = (targetPos - hrp.Position).Unit
                                hrp.AssemblyLinearVelocity = dir * currentSpeed

                                    hrp.CFrame = CFrame.new(hrp.Position, nearest.Position) * CFrame.Angles(math.rad(90), 0, 0)

                                    hrp.CFrame = CFrame.new(hrp.Position, nearest.Position)
                                end

                                    startedCollecting = true
                                end
